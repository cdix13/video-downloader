# Prevent sleep while running
caffeinate -i python3 app.py


# ============================================================================
# Linux - run_server_linux.sh
# ============================================================================
# Save this as: run_server_linux.sh
# Usage: chmod +x run_server_linux.sh && ./run_server_linux.sh

#!/bin/bash
echo "🎬 Starting Video Downloader Server (Linux)"
echo "🛡️  Preventing system sleep with systemd-inhibit..."
echo ""

# Check if systemd-inhibit is available
if command -v systemd-inhibit &> /dev/null; then
    systemd-inhibit --what=idle:sleep --who="Video Downloader" --why="Server is running" python3 app.py
else
    echo "⚠️  systemd-inhibit not found, running without sleep prevention"
    python3 app.py
fi


# ============================================================================
# Windows - run_server_windows.bat
# ============================================================================
# Save this as: run_server_windows.bat
# Usage: Double-click the file or run in Command Prompt

@echo off
echo ============================================================================
echo 🎬 Video Downloader Server - Windows
echo ============================================================================
echo.
echo 🛡️  Preventing system sleep...
echo.

REM Prevent sleep using powercfg
powercfg /change standby-timeout-ac 0
powercfg /change monitor-timeout-ac 0

echo ⚠️  Sleep mode disabled. Will restore on exit.
echo.
echo 🚀 Starting server...
echo.

REM Start Python server
python app.py

REM Restore power settings on exit
echo.
echo 🔄 Restoring power settings...
powercfg /change standby-timeout-ac 30
powercfg /change monitor-timeout-ac 10

echo.
echo ✅ Server stopped. Power settings restored.
pause


# ============================================================================
# Windows PowerShell - run_server_windows.ps1
# ============================================================================
# Save this as: run_server_windows.ps1
# Usage: Right-click -> Run with PowerShell

Write-Host "============================================================================" -ForegroundColor Cyan
Write-Host "🎬 Video Downloader Server - Windows PowerShell" -ForegroundColor Cyan
Write-Host "============================================================================" -ForegroundColor Cyan
Write-Host ""

# Prevent system from sleeping
Write-Host "🛡️  Preventing system sleep..." -ForegroundColor Yellow

# Using PowerShell to keep system awake
$code = @'
[DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)]
public static extern uint SetThreadExecutionState(uint esFlags);
'@

$ES_CONTINUOUS = [uint32]"0x80000000"
$ES_SYSTEM_REQUIRED = [uint32]"0x00000001"
$ES_DISPLAY_REQUIRED = [uint32]"0x00000002"

$type = Add-Type -MemberDefinition $code -Name System -Namespace Win32 -PassThru

# Prevent sleep
$null = $type::SetThreadExecutionState($ES_CONTINUOUS -bor $ES_SYSTEM_REQUIRED -bor $ES_DISPLAY_REQUIRED)

Write-Host "✅ Sleep prevention enabled" -ForegroundColor Green
Write-Host ""
Write-Host "🚀 Starting server..." -ForegroundColor Yellow
Write-Host ""

try {
    # Start Python server
    python app.py
}
finally {
    # Restore normal power state
    Write-Host ""
    Write-Host "🔄 Restoring power settings..." -ForegroundColor Yellow
    $null = $type::SetThreadExecutionState($ES_CONTINUOUS)
    Write-Host "✅ Power settings restored" -ForegroundColor Green
}

Write-Host ""
Write-Host "Press any key to exit..." -ForegroundColor Cyan
$null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")


# ============================================================================
# systemd Service (Linux) - video-downloader.service
# ============================================================================
# Save this as: /etc/systemd/system/video-downloader.service
# Enable: sudo systemctl enable video-downloader.service
# Start: sudo systemctl start video-downloader.service

[Unit]
Description=Video Downloader Web Service
After=network.target

[Service]
Type=simple
User=YOUR_USERNAME
WorkingDirectory=/path/to/video_downloader_web
ExecStart=/usr/bin/python3 /path/to/video_downloader_web/app.py
Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal

# Prevent system sleep
ExecStartPre=/usr/bin/systemd-inhibit --what=idle:sleep --who="Video Downloader" --why="Server is running"

[Install]
WantedBy=multi-user.target

python app.py\n\

#!/bin/bash

echo "🐳 Building Video Downloader with Tor..."

# Detect host IP
if [[ "$OSTYPE" == "darwin"* ]]; then
    HOST_IP=$(ipconfig getifaddr en0 || ipconfig getifaddr en1)
elif [[ "$OSTYPE" == "linux-gnu"* ]]; then
    HOST_IP=$(hostname -I | awk '{print $1}')
else
    HOST_IP="192.168.1.100"
fi

echo "🌐 Detected Host IP: $HOST_IP"

# Build image
docker build -t video-downloader .

# Stop old container
docker stop video-downloader 2>/dev/null
docker rm video-downloader 2>/dev/null

# Run new container
docker run -d \
  --name video-downloader \
  -p 5001:5001 \
  -e HOST_IP=$HOST_IP \
  -v $(pwd)/downloads:/app/downloads \
  -v $(pwd)/download_history.json:/app/download_history.json \
  --cap-add=NET_ADMIN \
  --restart unless-stopped \
  video-downloader

echo ""
echo "✅ Container started!"
echo "🧅 Tor is running inside the container"
echo ""
echo "📱 Access URLs:"
echo "   • Local:   http://localhost:5001"
echo "   • Network: http://$HOST_IP:5001"
echo ""
echo "🔍 View logs: docker logs -f video-downloader"
echo "🛑 Stop:      docker stop video-downloader"
echo ""
echo "💡 In the web UI, select proxy: 'Tor (port 9050)'"


# ============================================================================
# Advanced Dockerfile with Multiple Tor Instances
# ============================================================================
# Save this as: Dockerfile.multi-tor
# For high-performance parallel downloads through Tor

FROM python:3.11-slim

# Install dependencies
RUN apt-get update && apt-get install -y \
    ffmpeg \
    curl \
    tor \
    supervisor \
    && rm -rf /var/lib/apt/lists/*

# Configure multiple Tor instances
RUN mkdir -p /var/lib/tor-instances/tor{1,2,3}

# Create Tor configs for 3 instances
RUN for i in 1 2 3; do \
    port=$((9050 + i - 1)); \
    echo "SocksPort 0.0.0.0:$port" > /etc/tor/torrc$i; \
    echo "DataDirectory /var/lib/tor-instances/tor$i" >> /etc/tor/torrc$i; \
    echo "ControlPort $((9150 + i - 1))" >> /etc/tor/torrc$i; \
    done

# Set working directory
WORKDIR /app

# Copy requirements
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy application
COPY app.py .
COPY templates/ templates/

# Create directories
RUN mkdir -p downloads/videos downloads/audio

# Supervisor config for managing Tor instances
RUN echo '[supervisord]\n\
nodaemon=true\n\
\n\
[program:tor1]\n\
command=/usr/bin/tor -f /etc/tor/torrc1\n\
autorestart=true\n\
\n\
[program:tor2]\n\
command=/usr/bin/tor -f /etc/tor/torrc2\n\
autorestart=true\n\
\n\
[program:tor3]\n\
command=/usr/bin/tor -f /etc/tor/torrc3\n\
autorestart=true\n\
\n\
[program:app]\n\
command=python app.py\n\
directory=/app\n\
autorestart=true\n\
' > /etc/supervisor/conf.d/supervisord.conf

EXPOSE 5001

CMD ["/usr/bin/supervisord"]

# Linux/Mac - Auto-detect host IP and pass to Docker
# ---------------------------------------------------
# Save this as: docker-run.sh

#!/bin/bash

# Detect host IP
if [[ "$OSTYPE" == "darwin"* ]]; then
    # macOS
    HOST_IP=$(ipconfig getifaddr en0 || ipconfig getifaddr en1)
elif [[ "$OSTYPE" == "linux-gnu"* ]]; then
    # Linux
    HOST_IP=$(hostname -I | awk '{print $1}')
else
    # Fallback
    HOST_IP=$(ip route get 8.8.8.8 | awk -F"src " 'NR==1{split($2,a," ");print a[1]}')
fi

echo "🌐 Detected Host IP: $HOST_IP"

# Build and run with host IP
docker build -t video-downloader .
docker run -d \
  --name video-downloader \
  -p 5001:5001 \
  -e HOST_IP=$HOST_IP \
  -v $(pwd)/downloads:/app/downloads \
  -v $(pwd)/download_history.json:/app/download_history.json \
  --restart unless-stopped \
  video-downloader

echo "✅ Server started!"
echo "📱 Access at: http://$HOST_IP:5001"
